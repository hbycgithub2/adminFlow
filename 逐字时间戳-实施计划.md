```java
/**
 * 发送TTS请求并获取音频数据+逐字时间戳
 */
public class TTSResponseWithTimestamps {
    private byte[] audioData;
    private List<WordTimestamp> wordTimestamps;
}

public TTSResponseWithTimestamps sendTTSRequestWithTimestamps(String payload, String speaker) throws Exception {
    // 1. 发送请求
    // 2. 解析流式响应，提取音频数据和逐字时间戳
    // 3. 返回结果
    
    List<WordTimestamp> wordTimestamps = new ArrayList<>();
    ByteArrayOutputStream audioData = new ByteArrayOutputStream();
    
    // 逐行读取响应
    while ((line = reader.readLine()) != null) {
        // 解析JSON
        JSONObject json = new JSONObject(line);
        
        // 提取音频数据（code=0）
        if (json.getInt("code") == 0) {
            String data = json.getString("data");
            byte[] chunkAudio = Base64.getDecoder().decode(data);
            audioData.write(chunkAudio);
        }
        
        // ✅ 提取逐字时间戳（新增）
        if (json.has("word_timestamps")) {
            JSONArray words = json.getJSONArray("word_timestamps");
            for (int i = 0; i < words.length(); i++) {
                JSONObject word = words.getJSONObject(i);
                WordTimestamp timestamp = WordTimestamp.builder()
                        .word(word.getString("word"))
                        .startTime(word.getDouble("start_time") / 1000.0)  // 毫秒转秒
                        .endTime(word.getDouble("end_time") / 1000.0)
                        .build();
                wordTimestamps.add(timestamp);
            }
        }
    }
    
    return new TTSResponseWithTimestamps(audioData.toByteArray(), wordTimestamps);
}
```

### 第4步：创建字幕切分器（1小时）

#### 4.1 创建WordTimestampSplitter类

**文件：** `WordTimestampSplitter.java`

```java
package com.hmall.tts.volcengine.subtitle;

import com.hmall.tts.volcengine.docx.TextSegment;
import com.hmall.tts.volcengine.dto.WordTimestamp;
import com.hmall.tts.subtitle.dto.SubtitleSegment;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

/**
 * 逐字时间戳切分器
 * 根据原始文档切分合并后的逐字时间戳
 * 
 * @author Kiro
 * @since 2026-08-14
 */
@Slf4j
@Component
public class WordTimestampSplitter {
    
    /**
     * 切分逐字时间戳
     * 
     * @param originalSegments 原始文档片段（第1行、第3行）
     * @param mergedText TTS合并后的文本（"你好很高兴认识你"）
     * @param wordTimestamps TTS返回的逐字时间戳
     * @return 切分后的字幕片段
     */
    public List<SubtitleSegment> split(
        List<TextSegment> originalSegments,
        String mergedText,
        List<WordTimestamp> wordTimestamps
    ) {
        List<SubtitleSegment> subtitles = new ArrayList<>();
        
        if (originalSegments == null || originalSegments.isEmpty()) {
            log.warn("原始片段为空，无法切分");
            return subtitles;
        }
        
        if (wordTimestamps == null || wordTimestamps.isEmpty()) {
            log.warn("逐字时间戳为空，无法切分");
            return subtitles;
        }
        
        log.info("开始切分逐字时间戳，原始片段数：{}，逐字时间戳数：{}", 
                originalSegments.size(), wordTimestamps.size());
        
        int wordIndex = 0;  // 当前处理到第几个字
        
        for (int i = 0; i < originalSegments.size(); i++) {
            TextSegment originalSegment = originalSegments.get(i);
            String originalText = originalSegment.getText();  // "你好"
            int textLength = originalText.length();  // 2
            
            if (textLength == 0) {
                continue;  // 跳过空片段
            }
            
            // 提取这一行对应的逐字时间戳
            List<WordTimestamp> segmentWords = new ArrayList<>();
            for (int j = 0; j < textLength && wordIndex < wordTimestamps.size(); j++) {
                segmentWords.add(wordTimestamps.get(wordIndex));
                wordIndex++;
            }
            
            if (segmentWords.isEmpty()) {
                log.warn("片段{}没有对应的逐字时间戳，跳过", i);
                continue;
            }
            
            // 计算这一行的开始和结束时间
            double start = segmentWords.get(0).getStartTime();
            double end = segmentWords.get(segmentWords.size() - 1).getEndTime();
            
            // 创建字幕片段
            SubtitleSegment subtitle = SubtitleSegment.builder()
                    .id(i)
                    .text(originalText)
                    .startTime(start)
                    .duration(end - start)
                    .isBold(originalSegment.getIsBold())
                    .words(segmentWords)  // ← 保存逐字时间戳
                    .build();
            
            subtitles.add(subtitle);
            
            log.debug("切分字幕{}: 文本=[{}], 开始={:.2f}s, 结束={:.2f}s, 时长={:.2f}s", 
                    i, originalText, start, end, end - start);
        }
        
        log.info("切分完成，生成{}个字幕片段", subtitles.size());
        
        return subtitles;
    }
}
```

#### 5.2 新增buildDialogSegmentsWithTimestamps方法

```java
private List<DialogSegment> buildDialogSegmentsWithTimestamps(
    List<TextSegment> originalSegments,
    List<AudioSegmentWithTimestamps> audioSegments,
    VoiceConfig voiceConfig
) {
    List<DialogSegment> dialogSegments = new ArrayList<>();
    
    // 遍历每个音频片段（可能合并了多行）
    for (AudioSegmentWithTimestamps audioSegment : audioSegments) {
        MergedSegment mergedSegment = audioSegment.getMergedSegment();
        List<WordTimestamp> wordTimestamps = audioSegment.getWordTimestamps();
        
        // 获取这个音频片段对应的原始片段
        List<TextSegment> relatedOriginalSegments = mergedSegment.getOriginalSegments();
        
        // ✅ 使用WordTimestampSplitter切分
        List<SubtitleSegment> subtitles = wordTimestampSplitter.split(
            relatedOriginalSegments,
            mergedSegment.getText(),
            wordTimestamps
        );
        
        // 转换为DialogSegment
        for (SubtitleSegment subtitle : subtitles) {
            DialogSegment dialogSegment = DialogSegment.builder()
                    .text(subtitle.getText())
                    .isBold(subtitle.getIsBold())
                    .startTime(subtitle.getStartTime())
                    .duration(subtitle.getDuration())
                    .voiceId(mergedSegment.getSpeaker())
                    .charTimings(convertWordTimestampsToCharTimings(subtitle.getWords()))  // 转换为CharTiming
                    .build();
            
            dialogSegments.add(dialogSegment);
        }
    }
    
    return dialogSegments;
}
```
