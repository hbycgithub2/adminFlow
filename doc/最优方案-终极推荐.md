```java
/**
 * 智能判断：自动模式 vs 手动模式
 */
private GenerationMode determineMode(List<TextSegment> segments) {
    // 条件1：文本长度
    int totalLength = segments.stream()
            .mapToInt(s -> s.getText().length())
            .sum();
    
    if (totalLength > 800) {
        return GenerationMode.MANUAL;  // 超过800字符，必须分段
    }
    
    // 条件2：角色数量（关键判断）
    Set<String> speakers = segments.stream()
            .map(TextSegment::getSpeaker)
            .collect(Collectors.toSet());
    
    if (speakers.size() == 1) {
        return GenerationMode.AUTO;  // 单角色，用自动模式（最快）
    }
    
    // 条件3：角色切换频率
    int speakerChanges = 0;
    String lastSpeaker = "";
    for (TextSegment seg : segments) {
        if (!seg.getSpeaker().equals(lastSpeaker)) {
            speakerChanges++;
            lastSpeaker = seg.getSpeaker();
        }
    }
    
    if (speakerChanges <= 2) {
        return GenerationMode.AUTO;  // 角色切换少，用自动模式
    }
    
    // 条件4：段落与角色的关系
    Map<Integer, Set<String>> paragraphSpeakers = new HashMap<>();
    for (TextSegment seg : segments) {
        paragraphSpeakers.computeIfAbsent(seg.getParagraphId(), k -> new HashSet<>())
                .add(seg.getSpeaker());
    }
    
    boolean allParagraphsSingleSpeaker = paragraphSpeakers.values().stream()
            .allMatch(speakers -> speakers.size() == 1);
    
    if (allParagraphsSingleSpeaker) {
        return GenerationMode.MANUAL_OPTIMIZED;  // 段落内单角色，用优化手动模式
    }
    
    return GenerationMode.MANUAL_COMPLEX;  // 复杂场景，用完整手动模式
}
```

```java
/**
 * 优化手动模式：按paragraphId合并（修复bug）
 */
private AudioGenerationResult generateWithOptimizedManual(
        List<TextSegment> segments, 
        VoiceConfig voiceConfig) {
    
    // 1. 按paragraphId分组
    Map<Integer, List<TextSegment>> paragraphMap = segments.stream()
            .collect(Collectors.groupingBy(TextSegment::getParagraphId));
    
    List<MergedSegment> mergedSegments = new ArrayList<>();
    
    // 2. 每个段落生成一个MergedSegment
    for (Map.Entry<Integer, List<TextSegment>> entry : paragraphMap.entrySet()) {
        Integer paragraphId = entry.getKey();
        List<TextSegment> paragraphSegments = entry.getValue();
        
        // 合并文本
        String text = paragraphSegments.stream()
                .map(TextSegment::getText)
                .collect(Collectors.joining());
        
        // 选择speaker（使用第一个segment的speaker）
        String speaker = paragraphSegments.get(0).getSpeaker();
        
        MergedSegment merged = new MergedSegment();
        merged.setText(text);
        merged.setSpeaker(speaker);
        merged.setParagraphId(paragraphId);
        
        mergedSegments.add(merged);
    }
    
    // 3. 并发TTS
    List<AudioSegment> audioSegments = synthesizeParallel(mergedSegments, voiceConfig);
    
    // 4. WhisperX批量对齐
    List<List<CharTimestamp>> batchResults = alignBatch(audioSegments);
    
    // 5. 构建DialogSegment（简化版）
    List<DialogSegment> dialogSegments = buildDialogSegmentsSimple(
            mergedSegments, audioSegments, batchResults, voiceConfig);
    
    // 6. 合并音频
    byte[] finalAudio = audioMerger.merge(audioSegments, voiceConfig.getSampleRate());
    
    return new AudioGenerationResult(finalAudio, dialogSegments, totalDuration);
}

/**
 * 简化版DialogSegment构建（一一对应，无复杂匹配）
 */
private List<DialogSegment> buildDialogSegmentsSimple(
        List<MergedSegment> mergedSegments,
        List<AudioSegment> audioSegments,
        List<List<CharTimestamp>> batchResults,
        VoiceConfig voiceConfig) {
    
    List<DialogSegment> dialogSegments = new ArrayList<>();
    double currentTime = 0.0;
    
    // ✅ 关键：直接按索引一一对应（无需复杂匹配）
    for (int i = 0; i < mergedSegments.size(); i++) {
        MergedSegment merged = mergedSegments.get(i);
        AudioSegment audio = audioSegments.get(i);  // ← 一一对应
        List<CharTimestamp> whisperXChars = batchResults.get(i);  // ← 一一对应
        
        // 转换WhisperX结果
        List<CharTiming> charTimings = convertWhisperXToCharTimings(
                whisperXChars, currentTime);
        
        // 计算时长
        double duration = whisperXChars.get(whisperXChars.size() - 1).getEndTime();
        
        // 创建DialogSegment
        DialogSegment dialog = DialogSegment.builder()
                .index(i)
                .text(merged.getText())
                .voiceId(merged.getSpeaker())
                .startTime(currentTime)
                .duration(duration)
                .charTimings(charTimings)
                .build();
        
        dialogSegments.add(dialog);
        
        // 累加时间（音频时长 + 停顿）
        currentTime += duration;
        if (audio.getNeedPause()) {
            currentTime += audio.getPauseDuration() / 1000.0;
        }
    }
    
    return dialogSegments;
}
```

```java
/**

 */
private AudioGenerationResult generateWithComplexManual(
        List<TextSegment> segments, 
        VoiceConfig voiceConfig) {
    
    // 1. 按speaker合并（保留多音色）
    List<MergedSegment> mergedSegments = segmentMerger.merge(segments);
    
    // 2. 并发TTS
    List<AudioSegment> audioSegments = synthesizeParallel(mergedSegments, voiceConfig);
    
    // 3. WhisperX批量对齐
    List<List<CharTimestamp>> batchResults = alignBatch(audioSegments);
    
    // 4. 构建DialogSegment（智能匹配版）
    List<DialogSegment> dialogSegments = buildDialogSegmentsSmart(
            segments, audioSegments, batchResults, voiceConfig);
    
    // 5. 合并音频
    byte[] finalAudio = audioMerger.merge(audioSegments, voiceConfig.getSampleRate());
    
    return new AudioGenerationResult(finalAudio, dialogSegments, totalDuration);
}

/**
 * 智能版DialogSegment构建（按文本内容匹配）
 */
private List<DialogSegment> buildDialogSegmentsSmart(
        List<TextSegment> originalSegments,
        List<AudioSegment> audioSegments,
        List<List<CharTimestamp>> batchResults,
        VoiceConfig voiceConfig) {
    
    List<DialogSegment> dialogSegments = new ArrayList<>();
    
    // 1. 按paragraphId分组原始segments
    Map<Integer, List<TextSegment>> paragraphMap = originalSegments.stream()
            .collect(Collectors.groupingBy(TextSegment::getParagraphId));
    
    int audioIndex = 0;
    int batchIndex = 0;
    double currentTime = 0.0;
    
    // 2. 遍历每个段落
    for (Map.Entry<Integer, List<TextSegment>> entry : paragraphMap.entrySet()) {
        Integer paragraphId = entry.getKey();
        List<TextSegment> paragraphSegments = entry.getValue();
        
        // 合并段落文本
        String paragraphText = paragraphSegments.stream()
                .map(TextSegment::getText)
                .collect(Collectors.joining());
        
        // 3. 按文本内容匹配AudioSegment
        List<AudioSegment> matchedAudios = new ArrayList<>();
        List<List<CharTimestamp>> matchedBatches = new ArrayList<>();
        int textIndex = 0;
        
        while (textIndex < paragraphText.length() && audioIndex < audioSegments.size()) {
            AudioSegment audio = audioSegments.get(audioIndex);
            String audioText = audio.getMergedSegment().getText();
            
            // 检查文本是否匹配
            if (paragraphText.substring(textIndex).startsWith(audioText)) {
                matchedAudios.add(audio);
                matchedBatches.add(batchResults.get(batchIndex));
                textIndex += audioText.length();
                audioIndex++;
                batchIndex++;
            } else {
                log.error("文本匹配失败：期望「{}」，实际「{}」", 
                         paragraphText.substring(textIndex, 
                         Math.min(textIndex + 20, paragraphText.length())), 
                         audioText.substring(0, Math.min(20, audioText.length())));
                break;
            }
        }
        
        // 4. 合并匹配的WhisperX结果
        List<CharTiming> charTimings = new ArrayList<>();
        double paragraphDuration = 0.0;
        
        for (int i = 0; i < matchedAudios.size(); i++) {
            List<CharTimestamp> whisperXChars = matchedBatches.get(i);
            
            // 转换时间戳（加上currentTime偏移）
            for (CharTimestamp wchar : whisperXChars) {
                CharTiming ct = new CharTiming();
                ct.setCharacter(wchar.getCharacter());
                ct.setStartTime(currentTime + paragraphDuration + wchar.getStartTime());
                ct.setDuration(wchar.getDuration());
                charTimings.add(ct);
            }
            
            // 累加时长
            double segDuration = whisperXChars.get(whisperXChars.size() - 1).getEndTime();
            paragraphDuration += segDuration;
            
            // 加停顿
            AudioSegment audio = matchedAudios.get(i);
            if (audio.getNeedPause()) {
                paragraphDuration += audio.getPauseDuration() / 1000.0;
            }
        }
        
        // 5. 创建DialogSegment
        DialogSegment dialog = DialogSegment.builder()
                .index(dialogSegments.size())
                .text(paragraphText)
                .voiceId(paragraphSegments.get(0).getSpeaker())
                .startTime(currentTime)
                .duration(paragraphDuration)
                .charTimings(charTimings)
                .build();
        
        dialogSegments.add(dialog);
        currentTime += paragraphDuration;
    }
    
    return dialogSegments;
}
```
